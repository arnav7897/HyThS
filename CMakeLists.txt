cmake_minimum_required(VERSION 3.16)
project(hta_schnorr LANGUAGES CXX)

find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)

include(FetchContent)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(MCL_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(MCL_BUILD_SAMPLE OFF CACHE BOOL "" FORCE)
set(MCL_USE_GMP OFF CACHE BOOL "" FORCE)
FetchContent_Declare(mcl
  URL https://codeload.github.com/herumi/mcl/tar.gz/9f49ba06593e2d426ab6fb9b0e17bc125a6f720c
  URL_HASH SHA256=0a910a6c22935a73f537d3b1d95f08554ece7ccd5b2c04ff2e8ad4b19ca06de1)
FetchContent_MakeAvailable(mcl)
set_target_properties(mcl PROPERTIES EXCLUDE_FROM_ALL TRUE)

set(HTA_INSTRUMENTATION "none" CACHE STRING "Build instrumentation: none, asan-ubsan, tsan, coverage, fuzz")
set_property(CACHE HTA_INSTRUMENTATION PROPERTY STRINGS none asan-ubsan tsan coverage fuzz)
if(HTA_INSTRUMENTATION STREQUAL "asan-ubsan")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(HTA_INSTRUMENTATION STREQUAL "tsan")
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
elseif(HTA_INSTRUMENTATION STREQUAL "coverage")
  add_compile_options(--coverage)
  add_link_options(--coverage)
elseif(HTA_INSTRUMENTATION STREQUAL "fuzz")
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "The fuzz preset requires Clang")
  endif()
  add_compile_options(-fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=fuzzer-no-link,address,undefined)
elseif(NOT HTA_INSTRUMENTATION STREQUAL "none")
  message(FATAL_ERROR "Unknown HTA_INSTRUMENTATION: ${HTA_INSTRUMENTATION}")
endif()

add_library(hta_framing src/framing.cpp)
target_include_directories(hta_framing PUBLIC include)
target_compile_features(hta_framing PUBLIC cxx_std_17)

enable_testing()
add_executable(scaffold_smoke tests/scaffold_smoke.cpp)
add_test(NAME scaffold_smoke COMMAND scaffold_smoke)
add_executable(framing_test tests/framing_test.cpp)
target_link_libraries(framing_test PRIVATE hta_framing)
add_test(NAME framing COMMAND framing_test)

add_library(hta_parameters src/parameters.cpp)
target_include_directories(hta_parameters PUBLIC include)
target_compile_features(hta_parameters PUBLIC cxx_std_17)
add_executable(parameters_test tests/parameters_test.cpp)
target_link_libraries(parameters_test PRIVATE hta_parameters)
add_test(NAME parameters COMMAND parameters_test)

add_library(hta_expand_message src/expand_message.cpp)
target_include_directories(hta_expand_message PUBLIC include)
target_compile_features(hta_expand_message PUBLIC cxx_std_17)
target_link_libraries(hta_expand_message PRIVATE OpenSSL::Crypto)
add_executable(expand_message_test tests/expand_message_test.cpp)
target_link_libraries(expand_message_test PRIVATE hta_expand_message)
add_test(NAME expand_message COMMAND expand_message_test)

add_library(hta_scalar_challenge src/scalar_challenge.cpp)
target_include_directories(hta_scalar_challenge PUBLIC include)
target_compile_features(hta_scalar_challenge PUBLIC cxx_std_17)
target_link_libraries(hta_scalar_challenge PRIVATE hta_framing OpenSSL::Crypto)
add_executable(scalar_challenge_test tests/scalar_challenge_test.cpp)
target_link_libraries(scalar_challenge_test PRIVATE hta_scalar_challenge hta_framing)
add_test(NAME scalar_challenge COMMAND scalar_challenge_test)

add_library(hta_secp256k1_public src/secp256k1_public.cpp)
target_include_directories(hta_secp256k1_public PUBLIC include)
target_compile_features(hta_secp256k1_public PUBLIC cxx_std_17)
target_link_libraries(hta_secp256k1_public PRIVATE hta_framing mcl::mcl_st)
add_executable(secp256k1_public_test tests/secp256k1_public_test.cpp)
target_link_libraries(secp256k1_public_test PRIVATE hta_secp256k1_public)
add_test(NAME secp256k1_public COMMAND secp256k1_public_test)

add_library(hta_hash_to_field src/hash_to_field.cpp)
target_include_directories(hta_hash_to_field PUBLIC include)
target_compile_features(hta_hash_to_field PUBLIC cxx_std_17)
target_link_libraries(hta_hash_to_field PRIVATE hta_expand_message OpenSSL::Crypto)
